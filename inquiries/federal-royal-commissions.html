The Australian Parliamentary Handbook includes a searchable list of Royal Commissions with links to their final reports and responses by the government.

Lengths of Royal Commissions

Using the Handbook API you can determine the length of each Royal Commission based on the time elapsed between the Letters Patent establishing the Royal Commission, and the date of its final report.

Code 
df_rc = pd.read_json(
    "https://handbookapi.aph.gov.au/api/RoyalCommissions/RoyalCommissions",
    convert_dates=["DateLettersPatent", "DateFinalReportPresented"],
)
df_rc["days"] = (df_rc["DateFinalReportPresented"] - df_rc["DateLettersPatent"]).dt.days
# Wrap the text of long RC names for use in chart labels
df_rc["label"] = df_rc.apply(
    lambda x: textwrap.wrap(f"{x["Title"]} ({x["DateLettersPatent"].year})", 80), axis=1
)

slowest = df_rc.sort_values("days", ascending=False)[:10]
fastest = df_rc.sort_values("days", ascending=True)[:10]
chart_base = (
    alt.Chart()
    .mark_bar()
    .encode(
        y=alt.Y("label:N")
        .axis(labelLimit=0, labelBaseline="bottom")
        .sort("-x")
        .title(None),
        x=alt.X("days:Q")
        .scale(domain=(0, 1800))
        .title("number of days between letters patent and final report"),
        tooltip=[alt.Tooltip("label:N", title="name"), alt.Tooltip("days:Q", format=",")]
    )
    .properties(width="container", height=alt.Step(30), padding=20)
)

chart_slowest = chart_base.copy()
chart_slowest.data = slowest

chart_fastest = chart_base.copy()
chart_fastest.data = fastest
display(chart_slowest, chart_fastest)

Number of Royal Commissions by year and party

Code 
def find_party_by_date(df, date):
    govs = df.loc[(df["start_date"] <= date) & (df["end_date"] >= date)]
    if not govs.empty:
        return govs.iloc[0].to_dict()["party"]

df_gov = pd.read_csv("https://gist.github.com/wragge/bc74a0366faf41c89b31d505da7c56a4/raw/3d49b65ad238bd5c1c806234068d743df5305ef0/australian-federal-governments.csv", parse_dates=["start_date", "end_date"])

df_rc["party"] = df_rc["DateLettersPatent"].apply(
    lambda x: find_party_by_date(df_gov, x)
)

alt.Chart(df_rc).mark_bar().encode(
    x="year(DateLettersPatent):T", y="count():Q", color="party:N"
).properties(width="container")
